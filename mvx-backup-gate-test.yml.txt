name: MVX Backup Gate Test

on:
  workflow_dispatch:

permissions:
  contents: read

jobs:
  gate-test:
    runs-on: ubuntu-22.04
    timeout-minutes: 10

    env:
      CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
      CLOUDFLARE_ACCOUNT_ID: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
      MAIN_DB: housing-db
      WRANGLER_VERSION: 4.126.0

    steps:
      - name: Read automatic backup setting
        id: auto
        shell: bash
        run: |
          set -euo pipefail

          echo "Reading automatic backup setting from Main D1..."

          SETTING_JSON="$(
            npx --yes wrangler@"$WRANGLER_VERSION" d1 execute "$MAIN_DB" \
              --remote \
              --command "SELECT automatic_enabled FROM backup_settings WHERE id = 1;" \
              --json
          )"

          AUTOMATIC_ENABLED="$(
            printf '%s' "$SETTING_JSON" | node -e '
              let input = "";
              process.stdin.on("data", chunk => input += chunk);
              process.stdin.on("end", () => {
                const data = JSON.parse(input);
                const root = Array.isArray(data) ? data[0] : data;
                const row =
                  root?.results?.[0] ??
                  root?.result?.[0]?.results?.[0] ??
                  null;

                if (!row || !Object.prototype.hasOwnProperty.call(row, "automatic_enabled")) {
                  console.error("backup_settings row not found");
                  process.exit(2);
                }

                const enabled = Number(row.automatic_enabled) === 1;
                process.stdout.write(enabled ? "true" : "false");
              });
            '
          )"

          echo "automatic_enabled=$AUTOMATIC_ENABLED"

          if [ "$AUTOMATIC_ENABLED" = "true" ]; then
            echo "decision=run"
            echo "decision=run" >> "$GITHUB_OUTPUT"
          else
            echo "decision=skip"
            echo "decision=skip" >> "$GITHUB_OUTPUT"
          fi

      - name: Show simulated scheduled decision
        shell: bash
        env:
          DECISION: ${{ steps.auto.outputs.decision }}
        run: |
          set -euo pipefail

          if [ "$DECISION" = "run" ]; then
            echo "SIMULATED SCHEDULED RESULT: RUN BACKUP"
          elif [ "$DECISION" = "skip" ]; then
            echo "SIMULATED SCHEDULED RESULT: SKIP BACKUP"
          else
            echo "Unexpected gate decision." >&2
            exit 1
          fi
